%{
#include "tab.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
%}

%%

"array"             { return ARRAY; }
"of"                { return OF; }
"integer"           { return INTEGER; }
"real"              { return REAL; }
"byte"              { return BYTE; }
"word"              { return WORD; }
"char"              { return CHAR; }

":"                 { return COLON; }
"["                 { return LBRACKET; }
"]"                 { return RBRACKET; }
".."                { return DOTDOT; }
";"                 { return SEMICOLON; }

[A-Z]+              { yylval.id = strdup(yytext); return ROMAN; }

[a-zA-Z_][a-zA-Z0-9_]* { yylval.id = strdup(yytext); return ID; }

[ \t\n]             ;   /* игнорируем пробелы и переносы строк */
.                   { return yytext[0]; }

%%

int yywrap() {
    return 1;
}
